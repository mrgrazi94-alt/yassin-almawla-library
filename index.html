'flex' : 'none'; // تحديث الرقم داخل المربع البرتقالي الصغير var spans = bottomBar.querySelectorAll('span, div'); spans.forEach(function(el) { var txt = el.textContent.trim(); if (/^\d+$/.test(txt) && el.children.length === 0) { el.textContent = totalCount; } if (txt.includes('د.ع') && el.children.length === 0) { el.textContent = totalPrice.toLocaleString() + ' د.ع'; } }); // ربط زر "عرض السلة وإتمام الطلب" ليفتح نافذة السلة الأصلية var openCartBtn = bottomBar.querySelector('button'); if (openCartBtn && !openCartBtn.dataset.fixed) { openCartBtn.dataset.fixed = '1'; openCartBtn.onclick = function(e) { e.preventDefault(); // محاولة فتح السلة بكل الطرق الممكنة في موقعك var headerCartBtn = document.querySelector('#cartBtn, .cart-btn, [onclick*="cart"], [onclick*="Cart"]'); var cartModal = document.querySelector('#cartModal, .cart-modal, #cartSidebar, .cart-drawer'); if (headerCartBtn) { headerCartBtn.click(); } else if (typeof window.openCart === 'function') { window.openCart(); } else if (typeof window.toggleCart === 'function') { window.toggleCart(); } else if (cartModal) { cartModal.style.display = 'flex'; cartModal.classList.add('active', 'open'); } }; } } } // تشغيل المزامنة عند كل نقرة وفي الخلفية باستمرار document.addEventListener('click', function() { setTimeout(syncSmartBottomCart, 100); }); setInterval(syncSmartBottomCart, 500); })();
